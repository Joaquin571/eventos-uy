package contratos;

import java.lang.reflect.*;
import java.time.LocalDate;
import java.util.*;

/** Conversión estructural entre DTO locales y JAXB; no toma decisiones de dominio. */
public final class ConversorDatos {
    private ConversorDatos() { }
    @SuppressWarnings({"unchecked", "rawtypes"})
    public static <T> T convertir(Object origen, Class<T> destino) {
        if (origen == null) return null;
        if (destino == String.class && origen instanceof LocalDate) return (T) origen.toString();
        if (destino == LocalDate.class && origen instanceof String texto) return (T) LocalDate.parse(texto);
        if (destino.isPrimitive() || destino == String.class || Number.class.isAssignableFrom(destino)
                || destino == Boolean.class || destino == LocalDate.class) return (T) origen;
        if (destino == byte[].class) return (T) ((byte[])origen).clone();
        if (destino.isEnum()) return (T) Enum.valueOf((Class)destino, ((Enum<?>)origen).name());
        try {
            if (destino.getSimpleName().equals("DtUsuario"))
                destino=(Class<T>)Class.forName(destino.getPackageName()+"."+origen.getClass().getSimpleName());
            T resultado=destino.getConstructor().newInstance();
            for (Class<?> tipo=destino;tipo!=Object.class;tipo=tipo.getSuperclass()) {
                for (Field campo:tipo.getDeclaredFields()) {
                    if (Modifier.isStatic(campo.getModifiers()) || campo.isSynthetic()) continue;
                    String nombre=campo.getName();
                    // Credenciales solo viajan en EntradaUsuario; nunca en una salida/perfil/sesión.
                    if (tipo.getSimpleName().equals("DtUsuario")
                            && (nombre.equals("contrasenia") || nombre.equals("confirmacionContrasenia"))) continue;
                    String mayuscula=Character.toUpperCase(nombre.charAt(0))+nombre.substring(1);
                    Method getter=null;
                    for(String candidato:List.of("get"+mayuscula,"is"+mayuscula,nombre)) {
                        try { getter=origen.getClass().getMethod(candidato);break; }
                        catch(NoSuchMethodException ignorada) { }
                    }
                    if(getter==null) throw new NoSuchMethodException(nombre);
                    Object valor=getter.invoke(origen);
                    if(valor==null) continue;
                    campo.setAccessible(true);
                    if(valor instanceof Collection<?> coleccion) {
                        Class<?> elemento=(Class<?>)((ParameterizedType)campo.getGenericType()).getActualTypeArguments()[0];
                        List<Object> lista=new ArrayList<>();
                        for(Object item:coleccion) lista.add(convertir(item,elemento));
                        campo.set(resultado,lista);
                    } else campo.set(resultado,convertir(valor,campo.getType()));
                }
            }
            return resultado;
        } catch(ReflectiveOperationException | ClassCastException e) {
            throw new IllegalStateException("Los DTO de transporte no coinciden con el contrato.",e);
        }
    }
}

package central.seguridad;

import datatypes.Actor;
import datatypes.DtAsistente;
import datatypes.DtOrganizador;
import datatypes.DtUsuario;
import interfaces.ISistema;

import java.security.SecureRandom;
import java.time.Clock;
import java.time.Duration;
import java.time.Instant;
import java.util.Base64;
import java.util.Objects;
import java.util.concurrent.ConcurrentHashMap;
import java.util.regex.Pattern;

/** Registro en memoria del Central. La autorización de recursos sigue en ISistema. */
public final class SesionesRemotas {
    private static final String LOGIN_INVALIDO = "El nickname, el correo o la contraseña no son correctos.";
    private static final String SESION_INVALIDA = "La sesión remota no es válida.";
    private static final Pattern FORMATO = Pattern.compile("[A-Za-z0-9_-]{43}");
    private static final Base64.Encoder CODIFICADOR = Base64.getUrlEncoder().withoutPadding();

    private final ISistema sistema;
    private final SecureRandom aleatorio;
    private final Clock reloj;
    private final Duration duracion;
    private final ConcurrentHashMap<String, SesionRemota> sesiones = new ConcurrentHashMap<>();

    public SesionesRemotas(ISistema sistema, SecureRandom aleatorio, Clock reloj, Duration duracion) {
        this.sistema = Objects.requireNonNull(sistema);
        this.aleatorio = Objects.requireNonNull(aleatorio);
        this.reloj = Objects.requireNonNull(reloj);
        this.duracion = Objects.requireNonNull(duracion);
        if (duracion.isZero() || duracion.isNegative()) {
            throw new IllegalArgumentException("La duración de una sesión debe ser positiva.");
        }
    }

    public SesionEmitida iniciar(String identificador, String contrasenia) {
        DtUsuario usuario;
        try {
            usuario = sistema.autenticar(identificador, contrasenia);
        } catch (IllegalArgumentException error) {
            throw new IllegalArgumentException(LOGIN_INVALIDO);
        } catch (RuntimeException error) {
            throw new IllegalStateException("No se pudo completar la autenticación.");
        }
        if (!(usuario instanceof DtAsistente || usuario instanceof DtOrganizador)
                || usuario.getNickname() == null || usuario.getNickname().isBlank()) {
            throw new IllegalArgumentException(LOGIN_INVALIDO);
        }

        Actor actor = Actor.usuarioAutenticado(usuario);
        String nombre = usuario.getNombre();
        limpiarVencidas();
        Instant venceEn = reloj.instant().plus(duracion);
        var sesion = new SesionRemota(actor, venceEn);
        byte[] bytes = new byte[32];
        String credencial;
        do {
            aleatorio.nextBytes(bytes);
            credencial = CODIFICADOR.encodeToString(bytes);
        } while (sesiones.putIfAbsent(credencial, sesion) != null);

        return new SesionEmitida(credencial, actor, nombre, venceEn);
    }

    public Actor requerirActor(String credencial) {
        if (!formatoValido(credencial)) throw new IllegalArgumentException(SESION_INVALIDA);
        SesionRemota sesion = sesiones.get(credencial);
        if (sesion == null) throw new IllegalArgumentException(SESION_INVALIDA);
        if (!reloj.instant().isBefore(sesion.venceEn())) {
            sesiones.remove(credencial, sesion);
            throw new IllegalArgumentException(SESION_INVALIDA);
        }
        return sesion.actor();
    }

    /** Cierre idempotente: no informa si la referencia existía, venció o ya fue revocada. */
    public void revocar(String credencial) {
        if (formatoValido(credencial)) sesiones.remove(credencial);
    }

    /** Purga oportunista al emitir sesiones, además de la eliminación al consultar una vencida. */
    int limpiarVencidas() {
        Instant ahora = reloj.instant();
        int eliminadas = 0;
        for (var entrada : sesiones.entrySet()) {
            SesionRemota sesion = entrada.getValue();
            if (!ahora.isBefore(sesion.venceEn()) && sesiones.remove(entrada.getKey(), sesion)) {
                eliminadas++;
            }
        }
        return eliminadas;
    }

    private static boolean formatoValido(String credencial) {
        return credencial != null && credencial.length() == 43 && FORMATO.matcher(credencial).matches();
    }

    private record SesionRemota(Actor actor, Instant venceEn) { }

    /** Resultado interno de autenticación. La credencial se entrega solo al adaptador autorizado. */
    public record SesionEmitida(String credencial, Actor actor, String nombre, Instant venceEn) {
        @Override
        public String toString() {
            return "SesionEmitida[credencial omitida]";
        }
    }
}

final class CredencialesSoap {
    private static final Pattern BEARER=Pattern.compile("Bearer +([A-Za-z0-9_-]{43})",Pattern.CASE_INSENSITIVE);
    private CredencialesSoap() { }
    static String leerOpcional(MessageContext contexto) throws SesionSoapException {
        Object propiedad=contexto.get(MessageContext.HTTP_REQUEST_HEADERS);String valor=null;
        if(propiedad instanceof Map<?,?> encabezados) for(var entrada:encabezados.entrySet()) {
            if(!(entrada.getKey() instanceof String nombre) || !"Authorization".equalsIgnoreCase(nombre)) continue;
            if(valor!=null || !(entrada.getValue() instanceof List<?> valores) || valores.size()!=1
                    || !(valores.getFirst() instanceof String texto)) throw invalido();
            valor=texto;
        }
        if(valor==null) return null;
        var matcher=BEARER.matcher(valor);if(!matcher.matches()) throw invalido();
        return matcher.group(1);
    }
    static String leerRequerida(MessageContext contexto) throws SesionSoapException {
        String valor=leerOpcional(contexto);if(valor==null) throw invalido();return valor;
    }
    private static SesionSoapException invalido(){return new SesionSoapException("SOLICITUD_MALFORMADA","Se requiere un encabezado Authorization Bearer válido.");}

                EventoSoap dato = new EventoSoap();
                dato.setNombre(evento.getNombre());
                dato.setSigla(evento.getSigla());
                dato.setDescripcion(evento.getDescripcion());
                dato.setFechaAlta(evento.getFechaAlta() == null ? null : evento.getFechaAlta().toString());
                dato.setCategorias(evento.getCategorias() == null ? new ArrayList<>() : new ArrayList<>(evento.getCategorias()));
                resultado.add(dato);
            }
            return resultado;
        } catch (RuntimeException error) {
            LOG.log(Level.SEVERE, "No se pudo consultar el listado de eventos.", error);
            throw new ConsultaException("No se pudo consultar el listado de eventos. Intentá nuevamente más tarde.");

    @FunctionalInterface private interface Accion<T> { T ejecutar() throws Exception; }
    private <T> T invocar(Accion<T> accion) throws OperacionSoapException {
        try { return accion.ejecutar(); }
        catch(OperacionSoapException e) { throw e; }
        catch(excepciones.NoEncontradoException e) { throw new OperacionSoapException("NO_ENCONTRADO",e.getMessage()); }
        catch(excepciones.AccesoDenegadoException e) { throw new OperacionSoapException("ACCESO_DENEGADO",e.getMessage()); }
        catch(IllegalArgumentException e) { throw new OperacionSoapException("DATOS_INVALIDOS",e.getMessage()); }
        catch(Exception e) {
            if(e.getClass()==Exception.class) throw new OperacionSoapException("DATOS_INVALIDOS",e.getMessage());
            System.getLogger(getClass().getName()).log(System.Logger.Level.ERROR,"Fallo técnico del servicio Portal.");
            throw new OperacionSoapException("SERVICIO_NO_DISPONIBLE","No se pudo completar la operación.");
        }

        }

                    "El nickname, el correo o la contraseña no son correctos.");
        } catch (IllegalStateException e) {
            throw new SesionSoapException("SERVICIO_NO_DISPONIBLE", "No se pudo completar el inicio de sesión.");

                "Cache-Control", List.of("no-store")));

package central;

import central.seguridad.SesionesRemotas;
import central.soap.EventosPublicador;
import central.soap.SesionesPublicador;
import central.soap.PortalPublicador;
import com.sun.net.httpserver.HttpsConfigurator;
import com.sun.net.httpserver.HttpsServer;
import interfaces.ISistema;
import jakarta.xml.ws.Endpoint;

import java.io.IOException;
import java.net.InetSocketAddress;
import java.security.SecureRandom;
import java.time.Clock;
import java.util.ArrayList;
import java.util.List;
import java.util.Objects;
import java.util.concurrent.ExecutorService;
import java.util.concurrent.Executors;

/** Ciclo de vida del listener único y sus fachadas. La preparación TLS no inicializa JPA. */
public final class ServidorHttpsCentral implements AutoCloseable {
    private final ConfiguracionCentral configuracion;
    private final HttpsServer servidor;
    private final ExecutorService hilos;
    private final List<Endpoint> endpoints = new ArrayList<>();
    private boolean iniciado;
    private boolean cerrado;

    public ServidorHttpsCentral(ConfiguracionCentral configuracion) {
        this.configuracion = Objects.requireNonNull(configuracion);
        configuracion.validarHttps();
        var tls = configuracion.contextoTls();
        try {
            servidor = HttpsServer.create(new InetSocketAddress(configuracion.hostEscucha(), configuracion.puerto()), 0);
        } catch (IOException e) {
            throw new IllegalStateException("No se pudo abrir la dirección y el puerto HTTPS configurados.", e);
        }
        hilos = Executors.newVirtualThreadPerTaskExecutor();
        servidor.setHttpsConfigurator(new HttpsConfigurator(tls));
        servidor.setExecutor(hilos);
    }

    public synchronized void iniciar(ISistema sistema) {
        if (iniciado || cerrado) throw new IllegalStateException("El servidor HTTPS ya fue iniciado o cerrado.");
        Objects.requireNonNull(sistema);
        iniciado = true;
        try {
            var sesiones = new SesionesRemotas(sistema, new SecureRandom(), Clock.systemUTC(),
                    configuracion.duracionSesion());
            publicar(configuracion.rutaEventos(), new EventosPublicador(sistema));
            publicar(configuracion.rutaSesiones(), new SesionesPublicador(sesiones));
            publicar(configuracion.rutaPortal(), new PortalPublicador(sistema, sesiones));
            servidor.start();
        } catch (RuntimeException | Error e) {
            try { close(); } catch (RuntimeException cierre) { e.addSuppressed(cierre); }
            throw e;
        }
    }

    private void publicar(String ruta, Object fachada) {
        Endpoint endpoint = Endpoint.create(fachada);
        endpoints.add(endpoint);
        endpoint.publish(servidor.createContext(ruta));
    }

    public String urlPortal() { return configuracion.urlServicio(servidor.getAddress().getPort(), configuracion.rutaPortal()); }

    public String urlEventos() {
        return configuracion.urlServicio(servidor.getAddress().getPort(), configuracion.rutaEventos());
    }

    public String urlSesiones() {
        return configuracion.urlServicio(servidor.getAddress().getPort(), configuracion.rutaSesiones());
    }

    @Override
    public synchronized void close() {
        if (cerrado) return;
        cerrado = true;
        RuntimeException fallo = null;
        for (var endpoint : endpoints) {
            try {
                if (endpoint.isPublished()) endpoint.stop();
            } catch (RuntimeException e) {
                if (fallo == null) fallo = new IllegalStateException("Falló el cierre de un endpoint SOAP.");
                fallo.addSuppressed(e);
            }
        }
        try {
            servidor.stop(0);
        } finally {
            hilos.shutdownNow();
        }
        if (fallo != null) throw fallo;
    }
}

package central;

import java.io.IOException;
import java.nio.file.Files;
import java.nio.file.Path;
import java.net.URI;
import java.net.URISyntaxException;
import java.util.HashMap;
import java.util.Map;
import java.util.Properties;
import java.time.Duration;
import java.security.KeyStore;
import java.security.cert.X509Certificate;
import java.util.Arrays;
import javax.net.ssl.KeyManagerFactory;
import javax.net.ssl.SSLContext;

/** Configuración obligatoria; no utiliza credenciales ni destinos de respaldo. */
public final class ConfiguracionCentral {
    private final Properties propiedades = new Properties();

    public ConfiguracionCentral(Path archivo) {
        try (var lector = Files.newBufferedReader(archivo)) {
            propiedades.load(lector);
        } catch (IOException error) {
            throw new IllegalStateException("No se pudo leer la configuración del Servidor Central.", error);
        }
    }

    private String requerida(String clave) {
        String valor = propiedades.getProperty(clave);
        if (valor == null || valor.isBlank()) throw new IllegalStateException("Falta configurar " + clave);
        return valor.strip();
    }

    public String urlServicio() {
        return urlServicio(puerto(), rutaEventos());
    }

    public String urlSesiones() {
        return urlServicio(puerto(), rutaSesiones());
    }

    String urlServicio(int puerto, String ruta) {
        try {
            URI url = new URI("https", null, requerida("central.host"), puerto, ruta, null, null);
            if (url.getHost() == null) throw new IllegalArgumentException();
            return url.toString();
        } catch (IllegalArgumentException | URISyntaxException error) {
            throw new IllegalStateException("Revisá host, puerto y ruta SOAP en central.properties.", error);
        }
    }

    public int puerto() {
        try {
            int puerto = Integer.parseInt(requerida("central.puerto"));
            if (puerto < 0 || puerto > 65535) throw new NumberFormatException();
            return puerto;
        } catch (NumberFormatException e) {
            throw new IllegalStateException("central.puerto debe estar entre 0 y 65535.");
        }
    }

    public String hostEscucha() {
        return requerida("central.escucha");
    }

    public String rutaPortal() {
        String valor=propiedades.getProperty("central.ruta.portal","/portal");
        if(!valor.matches("/[A-Za-z0-9_-]+(?:/[A-Za-z0-9_-]+)*")) throw new IllegalStateException("Revisá central.ruta.portal.");
        return valor;
    }

    public String rutaEventos() { return ruta("central.ruta"); }
    public String rutaSesiones() { return ruta("central.ruta.sesiones"); }

    private String ruta(String propiedad) {
        String ruta = requerida(propiedad);
        if (!ruta.matches("/[A-Za-z0-9_-]+(?:/[A-Za-z0-9_-]+)*")) {
            throw new IllegalStateException("Revisá la ruta de " + propiedad + ".");
        }
        return ruta;
    }

    public Duration duracionSesion() {
        try {
            long segundos = Long.parseLong(requerida("sesiones.duracion.segundos"));
            if (segundos <= 0) throw new NumberFormatException();
            return Duration.ofSeconds(segundos);
        } catch (NumberFormatException e) {
            throw new IllegalStateException("sesiones.duracion.segundos debe ser un entero positivo.");
        }
    }

    public void validarHttps() {
        urlServicio();
        urlSesiones();
        hostEscucha();
        duracionSesion();
        String portal=rutaPortal();
        for(String otra:new String[]{rutaEventos(),rutaSesiones()})
            if(portal.equals(otra) || portal.startsWith(otra+"/") || otra.startsWith(portal+"/"))
                throw new IllegalStateException("La ruta del portal debe ser independiente.");
        String eventos = rutaEventos(), sesiones = rutaSesiones();
        if (eventos.equals(sesiones) || eventos.startsWith(sesiones + "/") || sesiones.startsWith(eventos + "/")) {
            throw new IllegalStateException("Las rutas de eventos y sesiones deben ser independientes.");
        }
    }

    public SSLContext contextoTls() {
        char[] claveAlmacen = clave("tls.almacen.password");
        char[] clavePrivada = null;
        try {
            clavePrivada = clave("tls.clave.password");
            Path archivo = Path.of(requerida("tls.almacen"));
            if (!archivo.isAbsolute()) {
                throw new IllegalStateException("tls.almacen debe ser una ruta absoluta.");
            }
            var almacen = KeyStore.getInstance("PKCS12");
            try (var entrada = Files.newInputStream(archivo)) {
                almacen.load(entrada, claveAlmacen);
            }
            int claves = 0;
            for (var alias = almacen.aliases(); alias.hasMoreElements();) {
                String nombre = alias.nextElement();
                if (!almacen.isKeyEntry(nombre)) continue;
                claves++;
                var cadena = almacen.getCertificateChain(nombre);
                if (cadena == null || cadena.length == 0) {
                    throw new IllegalStateException("El almacén TLS requiere una cadena de certificados.");
                }
                for (var certificado : cadena) ((X509Certificate) certificado).checkValidity();
            }
            if (claves != 1) throw new IllegalStateException("El PKCS#12 debe contener una única clave privada.");
            var gestor = KeyManagerFactory.getInstance(KeyManagerFactory.getDefaultAlgorithm());
            gestor.init(almacen, clavePrivada);
            var contexto = SSLContext.getInstance("TLS");
            contexto.init(gestor.getKeyManagers(), null, null);
            return contexto;
        } catch (IllegalStateException e) {
            throw e;
        } catch (Exception e) {
            throw new IllegalStateException("No se pudo cargar TLS. Revisá el PKCS#12, sus claves y certificados.");
        } finally {
            Arrays.fill(claveAlmacen, '\0');
            if (clavePrivada != null) Arrays.fill(clavePrivada, '\0');
        }
    }

    private char[] clave(String propiedad) {
        String clave = propiedades.getProperty(propiedad);
        if (clave == null || clave.isEmpty()) throw new IllegalStateException("Falta configurar " + propiedad);
        return clave.toCharArray();
    }

    public Map<String, String> propiedadesJpa() {
        String url = requerida("db.url");
        if (!url.startsWith("jdbc:postgresql://")) throw new IllegalStateException("db.url debe ser PostgreSQL.");
        Map<String, String> resultado = new HashMap<>();
        resultado.put("jakarta.persistence.jdbc.driver", "org.postgresql.Driver");
        resultado.put("jakarta.persistence.jdbc.url", url);
        resultado.put("jakarta.persistence.jdbc.user", requerida("db.usuario"));
        String clave = propiedades.getProperty("db.password");
        if (clave == null) throw new IllegalStateException("Falta configurar db.password");
        resultado.put("jakarta.persistence.jdbc.password", clave);
        // Esta fase no crea ni migra tablas. Requiere un esquema ya existente.
        resultado.put("hibernate.hbm2ddl.auto", "validate");
        return resultado;
    }
}
