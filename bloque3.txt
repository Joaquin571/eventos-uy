package web;

import jakarta.servlet.http.HttpServletRequest;
import jakarta.servlet.http.HttpServletResponse;
import jakarta.servlet.http.HttpSession;

import java.io.IOException;
import java.net.URLDecoder;
import java.nio.charset.StandardCharsets;
import java.security.MessageDigest;
import java.security.SecureRandom;
import java.util.Base64;

/** Token aleatorio por sesión. Nunca se registra ni se transporta en una URL. */
final class Csrf {
    static final String PARAMETRO = "_csrf";
    private static final String ATRIBUTO = Csrf.class.getName() + ".token";
    private static final SecureRandom ALEATORIO = new SecureRandom();

    private Csrf() { }

    static boolean validarPost(HttpServletRequest req, HttpServletResponse resp) throws IOException {
        HttpSession sesion = req.getSession(false);
        Object esperado = sesion == null ? null : sesion.getAttribute(ATRIBUTO);
        if (!(esperado instanceof String token) || tokenEnUrl(req.getQueryString())) return rechazar(resp);
        // Tomcat parsea también multipart cuando el servlet destino tiene @MultipartConfig.
        // No se usa getInputStream/getReader y el servlet conserva acceso a sus Part.
        String[] recibidos;
        try {
            recibidos = req.getParameterValues(PARAMETRO);
        } catch (IllegalStateException e) {
            // Tomcat aplica los límites multipart al parsear los parámetros, antes del Servlet.
            resp.sendError(HttpServletResponse.SC_REQUEST_ENTITY_TOO_LARGE,
                    "El formulario o la imagen supera el tamaño permitido.");
            return false;
        }
        if (recibidos == null || recibidos.length != 1 || recibidos[0] == null
                || recibidos[0].length() != token.length()
                || !MessageDigest.isEqual(token.getBytes(StandardCharsets.UTF_8),
                        recibidos[0].getBytes(StandardCharsets.UTF_8))) return rechazar(resp);
        req.setAttribute("csrfToken", token);
        return true;
    }

    static void prepararFormulario(HttpServletRequest req) {
        HttpSession sesion = req.getSession(true);
        String token;
        synchronized (sesion) {
            Object existente = sesion.getAttribute(ATRIBUTO);
            if (existente instanceof String valor) {
                token = valor;
            } else {
                byte[] bytes = new byte[32]; ALEATORIO.nextBytes(bytes);
                token = Base64.getUrlEncoder().withoutPadding().encodeToString(bytes);
                sesion.setAttribute(ATRIBUTO, token);
            }
        }
        req.setAttribute("csrfToken", token);
    }

    private static boolean tokenEnUrl(String query) {
        if (query == null) return false;
        try {
            for (String par : query.split("&")) {
                String nombre = URLDecoder.decode(par.split("=", 2)[0], StandardCharsets.UTF_8);
                if (PARAMETRO.equals(nombre)) return true;
            }
            return false;
        } catch (IllegalArgumentException e) {
            return true;
        }
    }

    private static boolean rechazar(HttpServletResponse resp) throws IOException {
        resp.sendError(HttpServletResponse.SC_FORBIDDEN,
                "El formulario expiró o no es válido. Volvé a abrirlo e intentá nuevamente.");
        return false;
    }
}

package web;

import jakarta.servlet.http.HttpServletRequest;

/** Mensajes de una sola lectura, separados por pantalla y sin parámetros en la URL. */
final class MensajesFlash {
    private record Mensaje(String exito, String advertencia) { }

    private MensajesFlash() { }

    static void guardar(HttpServletRequest req, String pantalla, String exito, String advertencia) {
        req.getSession().setAttribute(MensajesFlash.class.getName() + pantalla, new Mensaje(exito, advertencia));
    }

    static void mostrar(HttpServletRequest req, String pantalla) {
        if (!"GET".equals(req.getMethod())) return;
        var sesion = req.getSession(false);
        if (sesion == null) return;
        String clave = MensajesFlash.class.getName() + pantalla;
        Object valor;
        synchronized (sesion) {
            valor = sesion.getAttribute(clave);
            sesion.removeAttribute(clave);
        }
        if (valor instanceof Mensaje mensaje) {
            req.setAttribute("exito", mensaje.exito());
            req.setAttribute("advertencia", mensaje.advertencia());
        }
    }
}

package web;
import contratos.dto.*;
import jakarta.servlet.*;
import jakarta.servlet.annotation.WebServlet;
import jakarta.servlet.http.*;
import java.io.IOException;
import java.time.LocalDate;
import java.time.format.DateTimeFormatter;
import java.util.*;

@WebServlet("/consulta-registro")
public class ConsultaRegistroServlet extends HttpServlet {
    protected void doGet(HttpServletRequest req,HttpServletResponse resp)throws ServletException,IOException {
        String edicion=AccesoEdiciones.texto(req,"edicion"),asistente=AccesoEdiciones.texto(req,"asistente");
        var consulta=ContextoListener.central(req).consultarRegistros(edicion,asistente);
        Map<String,String> nombres=new HashMap<>();
        for(var usuario:consulta.getAsistentes())nombres.put(usuario.getNickname(),nombreCompleto(usuario));
        var propio=SesionWeb.usuario(req);if(propio!=null)nombres.put(propio.getNickname(),propio.getNombre());
        req.setAttribute("esOrganizador",consulta.isOrganizador());req.setAttribute("ediciones",consulta.getEdiciones());
        req.setAttribute("edicionElegida",edicion);req.setAttribute("asistenteElegido",asistente);
        req.setAttribute("registrado","1".equals(req.getParameter("registrado")));
        req.setAttribute("registros",consulta.getRegistros().stream().map(r->DetalleRegistroVista.de(r,nombres.getOrDefault(r.getNicknameAsistente(),r.getNicknameAsistente()))).toList());
        var detalle=consulta.getDetalle();
        if(detalle!=null)req.setAttribute("detalle",DetalleRegistroVista.de(detalle,nombres.getOrDefault(detalle.getNicknameAsistente(),detalle.getNicknameAsistente())));
        req.getRequestDispatcher("/WEB-INF/views/consulta-registro.jsp").forward(req,resp);
    }
    private String nombreCompleto(DtUsuario u){return u instanceof DtAsistente a?u.getNombre()+" "+a.getApellido():u.getNombre();}
}

package web;

import contratos.dto.DtRegistro;

import java.time.format.DateTimeFormatter;

/**
 * Registro listo para mostrar en la consulta de registro: la fecha y el costo ya vienen
 * formateados, y "usoCodigo" es "Sí" o "No". El código de patrocinio nunca llega a la vista.
 */
public class DetalleRegistroVista {

    private static final DateTimeFormatter FECHA = DateTimeFormatter.ofPattern("dd/MM/yyyy");

    private final String nickname;
    private final String nombreAsistente;
    private final String nombreEdicion;
    private final String nombreTipoRegistro;
    private final String fecha;
    private final String costo;
    private final String usoCodigo;

    public DetalleRegistroVista(String nickname, String nombreAsistente, String nombreEdicion,
                                String nombreTipoRegistro, String fecha, String costo,
                                String usoCodigo) {
        this.nickname = nickname;
        this.nombreAsistente = nombreAsistente;
        this.nombreEdicion = nombreEdicion;
        this.nombreTipoRegistro = nombreTipoRegistro;
        this.fecha = fecha;
        this.costo = costo;
        this.usoCodigo = usoCodigo;
    }

    /** Arma la vista a partir del DtRegistro que devuelve la lógica. */
    public static DetalleRegistroVista de(DtRegistro registro, String nombreAsistente) {
        return new DetalleRegistroVista(
                registro.getNicknameAsistente(),
                nombreAsistente,
                registro.getNombreEdicion(),
                registro.getNombreTipoRegistro(),
                registro.getFechaRegistro() == null ? "" : registro.getFechaRegistro().format(FECHA),
                formatearCosto(registro.getCosto()),
                registro.isUsoCodigo() ? "Sí" : "No");
    }

    /** 1500 se muestra como "1500" y 800.5 como "800.50". */
    public static String formatearCosto(float costo) {
        return costo == Math.rint(costo) ? String.valueOf((long) costo) : String.format("%.2f", costo);
    }

    public String getNickname() {
        return nickname;
    }

    public String getNombreAsistente() {
        return nombreAsistente;
    }

    public String getNombreEdicion() {
        return nombreEdicion;
    }

    public String getNombreTipoRegistro() {
        return nombreTipoRegistro;
    }

    public String getFecha() {
        return fecha;
    }

    public String getCosto() {
        return costo;
    }

    public String getUsoCodigo() {
        return usoCodigo;
    }
}

import jakarta.servlet.http.*;
import java.io.IOException;
import java.net.URLEncoder;
import java.nio.charset.StandardCharsets;
import java.time.LocalDate;
import java.time.format.DateTimeFormatter;
import java.util.*;

@WebServlet("/disponibilidad-usuario")
public class DisponibilidadUsuarioServlet extends HttpServlet {
    protected void doGet(HttpServletRequest req, HttpServletResponse resp) throws IOException {
        resp.setContentType("application/json;charset=UTF-8");
        resp.setHeader("Cache-Control", "no-store");
        String campo = req.getParameter("campo");
        String valor = req.getParameter("valor");
        boolean disponible;
        if (!("nickname".equals(campo) || "correo".equals(campo))) {
            resp.setStatus(400);
            resp.getWriter().write("{\"error\":\"Campo inválido\"}");
            return;
        }
        try {
            disponible = "nickname".equals(campo)
                    ? ContextoListener.central(req).disponibleNickname(valor)
                    : ContextoListener.central(req).disponibleCorreo(valor);
        } catch (web.remoto.ErrorCentral error) {
            resp.setStatus(HttpServletResponse.SC_SERVICE_UNAVAILABLE);
            resp.getWriter().write("{\"error\":\"Servicio no disponible\"}");
            return;
        }
        resp.getWriter().write("{\"disponible\":" + disponible + "}");
    }
}

package web;
import contratos.dto.*;
import jakarta.servlet.*;
import jakarta.servlet.annotation.WebServlet;
import jakarta.servlet.http.*;
import java.io.IOException;
import java.time.LocalDate;
import java.time.format.DateTimeFormatter;
import java.util.*;

import java.net.URLEncoder;
import java.nio.charset.StandardCharsets;
@WebServlet("/registro-edicion")
public class RegistroEdicionServlet extends HttpServlet {
    private static final DateTimeFormatter FECHA=DateTimeFormatter.ofPattern("dd/MM/yyyy");
    protected void doGet(HttpServletRequest req,HttpServletResponse resp)throws ServletException,IOException {
        mostrar(req,resp,AccesoEdiciones.texto(req,"evento"),AccesoEdiciones.texto(req,"edicion"),AccesoEdiciones.texto(req,"tipo"),"general","",null);
    }
    protected void doPost(HttpServletRequest req,HttpServletResponse resp)throws ServletException,IOException {
        String evento=AccesoEdiciones.texto(req,"evento"),edicion=AccesoEdiciones.texto(req,"edicion"),tipo=AccesoEdiciones.texto(req,"tipo");
        String modalidad=AccesoEdiciones.texto(req,"modalidad"),codigo=AccesoEdiciones.texto(req,"codigo");
        try {
            ContextoListener.central(req).registrar(edicion,tipo,modalidad,codigo);
            resp.sendRedirect(req.getContextPath()+"/consulta-registro?registrado=1&edicion="+URLEncoder.encode(edicion,StandardCharsets.UTF_8));
        }catch(IllegalArgumentException e){mostrar(req,resp,evento,edicion,tipo,modalidad,codigo,e.getMessage());}
    }
    private void mostrar(HttpServletRequest req,HttpServletResponse resp,String evento,String edicion,String tipo,String modalidad,String codigo,String error)throws ServletException,IOException {
        var consulta=ContextoListener.central(req).formularioRegistro(evento,edicion,tipo);
        req.setAttribute("eventos",consulta.getEventos());req.setAttribute("eventoElegido",consulta.getEvento());
        req.setAttribute("ediciones",consulta.getEdiciones());req.setAttribute("tipos",consulta.getTipos());
        var elegida=consulta.getEdicion();req.setAttribute("detalle",elegida);
        req.setAttribute("edicionElegida",elegida==null?"":elegida.getIdNombre());
        req.setAttribute("tipoElegido",consulta.getTipo()==null?"":consulta.getTipo().getIdNombre());
        Map<String,String> costos=new LinkedHashMap<>();for(var t:consulta.getTipos())costos.put(t.getIdNombre(),DetalleRegistroVista.formatearCosto(t.getCosto()));
        req.setAttribute("costos",costos);req.setAttribute("modalidad",modalidad);req.setAttribute("codigo",codigo);req.setAttribute("error",error);
        if(elegida!=null){req.setAttribute("fechaInicio",fecha(elegida.getFechaInicio()));req.setAttribute("fechaFin",fecha(elegida.getFechaFin()));}
        req.getRequestDispatcher("/WEB-INF/views/registro-edicion.jsp").forward(req,resp);
    }
    private String fecha(LocalDate valor){return valor==null?"":valor.format(FECHA);}
}

package web;
import contratos.ServicioPortal;
import jakarta.servlet.*;
import jakarta.servlet.annotation.WebListener;
import jakarta.servlet.http.HttpServletRequest;
import web.remoto.*;
/** Tomcat solamente crea clientes: no inicia persistencia ni una instancia de Sistema. */
@WebListener
public class ContextoListener implements ServletContextListener {
    public void contextInitialized(ServletContextEvent evento) {
        evento.getServletContext().setAttribute("configuracionCentral", ConfiguracionWeb.cargar());
    }
    static ConfiguracionWeb configuracion(HttpServletRequest req) {
        return (ConfiguracionWeb)req.getServletContext().getAttribute("configuracionCentral");
    }
    public static ServicioPortal central(HttpServletRequest req) {
        Object doble=req.getServletContext().getAttribute("central");
        if(doble instanceof ServicioPortal remoto)return remoto;
        return new ClienteCentral(configuracion(req).portal(), SesionWeb.credencial(req));
    }
    static ServicioSesiones sesiones(HttpServletRequest req) {
        Object doble=req.getServletContext().getAttribute("sesionesCentral");
        if(doble instanceof ServicioSesiones remoto)return remoto;
        return new ClienteSesionesCentral(configuracion(req).sesiones());
    }
}

package web;
import contratos.dto.*;
import jakarta.servlet.*;
import jakarta.servlet.annotation.WebServlet;
import jakarta.servlet.http.*;
import java.io.IOException;
@WebServlet("/login")
public class LoginServlet extends HttpServlet {
    protected void doGet(HttpServletRequest req,HttpServletResponse resp)throws ServletException,IOException {
        req.getRequestDispatcher("/WEB-INF/views/login.jsp").forward(req,resp);
    }
    protected void doPost(HttpServletRequest req,HttpServletResponse resp)throws ServletException,IOException {
        String identificador=req.getParameter("identificador");
        try {
            var inicio=ContextoListener.sesiones(req).iniciar(identificador,req.getParameter("contrasenia"));
            var anterior=req.getSession(false);if(anterior!=null)anterior.invalidate();
            var sesion=req.getSession(true);
            DtUsuario usuario="ORGANIZADOR".equals(inicio.rol())?new DtOrganizador():new DtAsistente();
            usuario.setNickname(inicio.nickname());usuario.setNombre(inicio.nombre());
            SesionWeb.guardar(sesion,usuario);sesion.setAttribute(SesionWeb.CREDENCIAL,inicio.credencial());
            resp.sendRedirect(req.getContextPath()+"/");
        }catch(IllegalArgumentException e){req.setAttribute("error",e.getMessage());req.setAttribute("identificador",identificador);doGet(req,resp);}
    }
}

package web;

import contratos.dto.DtAsistente;
import contratos.dto.DtOrganizador;
import contratos.dto.DtUsuario;
import jakarta.servlet.ServletException;
import jakarta.servlet.annotation.WebServlet;
import jakarta.servlet.http.HttpServlet;
import jakarta.servlet.http.HttpServletRequest;
import jakarta.servlet.http.HttpServletResponse;

import java.io.IOException;
import java.util.ArrayList;
import java.util.List;

@WebServlet("/usuarios")
public class UsuariosServlet extends HttpServlet {

    @Override
    protected void doGet(HttpServletRequest req, HttpServletResponse resp)
            throws ServletException, IOException {

        List<DtAsistente> asistentes = new ArrayList<>();
        List<DtOrganizador> organizadores = new ArrayList<>();

        for (DtUsuario usuario : ContextoListener.central(req).listarUsuarios()) {
            if (usuario instanceof DtAsistente asistente) {
                asistentes.add(asistente);
            } else if (usuario instanceof DtOrganizador organizador) {
                organizadores.add(organizador);
            }
        }

        req.setAttribute("asistentes", asistentes);
        req.setAttribute("organizadores", organizadores);
        req.getRequestDispatcher("/WEB-INF/views/usuarios.jsp").forward(req, resp);
    }
}

<file path="estacion-trabajo/src/main/java/swing/AceptarRechazarEdicionPanel.java">
package swing;

import datatypes.Actor;

import datatypes.DtEdicion;
import datatypes.DtEvento;
import datatypes.EstadoEdicion;
import implementacion.Fabrica;
import interfaces.ISistema;

import javax.swing.*;
import java.awt.*;
import java.time.format.DateTimeFormatter;
import java.util.LinkedHashMap;
import java.util.Map;

/**
 * Caso de uso Aceptar/Rechazar Edición de Evento (administrador).
 * Lista los eventos, luego las ediciones en estado Ingresada del evento elegido,
 * y permite aceptar o rechazar la seleccionada.
 */
public class AceptarRechazarEdicionPanel {

    private static final String SELECCIONE_EVENTO = "Seleccione un evento";
    private static final DateTimeFormatter FECHA = DateTimeFormatter.ofPattern("dd/MM/yyyy");

    private final transient ISistema sistema;

    private JPanel mainPanel;
    private JComboBox<String> comboEvento;
    private DefaultListModel<String> modeloEdiciones;
    private JList<String> listaEdiciones;
    private JTextArea txtDetalle;
    private JButton btnAceptar;
    private JButton btnRechazar;
    private JButton btnCerrar;

    // Ediciones Ingresada del evento elegido, por nombre
    private final Map<String, DtEdicion> ediciones = new LinkedHashMap<>();

    private boolean cargando = false;
    private Runnable accionCerrar = () -> {};

    private final Actor actor;

    public AceptarRechazarEdicionPanel(Actor actor) {
        this(actor, Fabrica.getInstance().getISistema());
    }

    AceptarRechazarEdicionPanel(Actor actor, ISistema sistema) {
        this.actor = java.util.Objects.requireNonNull(actor, "Se requiere Actor.");
        this.sistema = java.util.Objects.requireNonNull(sistema);

        inicializarComponentes();
        cargarEventos();

        comboEvento.addActionListener(e -> {
            if (!cargando) {
                cargarEdiciones();
            }
        });

        listaEdiciones.addListSelectionListener(e -> {
            if (!e.getValueIsAdjusting()) {
                mostrarDetalle();
            }
        });

        btnAceptar.addActionListener(e -> cambiarEstado(true));
        btnRechazar.addActionListener(e -> cambiarEstado(false));
        btnCerrar.addActionListener(e -> accionCerrar.run());
    }

    public JPanel getMainPanel() {
        return mainPanel;
    }

    public void setAccionCerrar(Runnable accionCerrar) {
        this.accionCerrar = accionCerrar;
    }

    public void refrescarDatos() {
        cargarEventos();
    }

    public void cargarEventos() {
        cargando = true;

        comboEvento.removeAllItems();
        comboEvento.addItem(SELECCIONE_EVENTO);

        for (DtEvento evento : sistema.listarEventos()) {
            comboEvento.addItem(evento.getNombre());
        }

        comboEvento.setSelectedIndex(0);
        cargando = false;

        cargarEdiciones();
    }

    private String eventoElegido() {
        Object seleccionado = comboEvento.getSelectedItem();

        if (seleccionado == null || SELECCIONE_EVENTO.equals(seleccionado)) {
            return null;
        }
        return seleccionado.toString();
    }

    private void cargarEdiciones() {
        ediciones.clear();
        modeloEdiciones.clear();

        String evento = eventoElegido();

        if (evento != null) {
            for (DtEdicion edicion : sistema.obtenerEdicionesEvento(evento, EstadoEdicion.INGRESADA)) {
                ediciones.put(edicion.getIdNombre(), edicion);
                modeloEdiciones.addElement(edicion.getIdNombre());
            }
        }

        if (evento == null) {
            txtDetalle.setText("Elija un evento para ver sus ediciones en estado Ingresada.");
        } else if (ediciones.isEmpty()) {
            txtDetalle.setText("El evento no tiene ediciones en estado Ingresada.");
        } else {
            txtDetalle.setText("Elija una edición para ver sus datos.");
        }

        actualizarBotones();
    }

    private void mostrarDetalle() {
        String nombre = listaEdiciones.getSelectedValue();

    }

    private void cambiarEstado(boolean aceptar) {
        String nombre = listaEdiciones.getSelectedValue();

        if (nombre == null) {
            return;
        }

        String accion = aceptar ? "aceptar" : "rechazar";

        int respuesta = JOptionPane.showConfirmDialog(mainPanel,
                "¿Desea " + accion + " la edición '" + nombre + "'?\n"
                        + "Esta decisión no se puede deshacer.",
                "Aceptar/Rechazar Edición", JOptionPane.YES_NO_OPTION);

        if (respuesta != JOptionPane.YES_OPTION) {
            return;
        }

        try {
            sistema.aceptarRechazarEdicion(nombre, aceptar, actor);

            JOptionPane.showMessageDialog(mainPanel,
                    "La edición '" + nombre + "' fue " + (aceptar ? "aceptada." : "rechazada."),

package web;

import contratos.dto.DtImagen;
import contratos.dto.TipoEntidadImagen;
import jakarta.servlet.annotation.WebServlet;
import jakarta.servlet.http.HttpServlet;
import jakarta.servlet.http.HttpServletRequest;
import jakarta.servlet.http.HttpServletResponse;

import java.io.IOException;

@WebServlet("/imagen")
public class ImagenServlet extends HttpServlet {

    @Override
    protected void doGet(HttpServletRequest req, HttpServletResponse resp) throws IOException {

        TipoEntidadImagen tipo = tipoDe(req.getParameter("tipo"));
        String id = req.getParameter("id");

        resp.setHeader("Cache-Control", "private, no-store");
        DtImagen imagen = (tipo == null)
                ? null
                : ContextoListener.central(req).obtenerImagen(tipo, id);

        if (imagen == null) {
            resp.sendRedirect(req.getContextPath() + "/img/default-avatar.png");
            return;
        }

        resp.setContentType(imagen.getTipoContenido());
        if (tipo != TipoEntidadImagen.EDICION) resp.setHeader("Cache-Control", "no-cache");
        resp.setHeader("X-Content-Type-Options", "nosniff");

        resp.setContentLength(imagen.getContenido().length);
        resp.getOutputStream().write(imagen.getContenido());
    }

    private TipoEntidadImagen tipoDe(String valor) {
        if (valor == null) {
            return null;
        }

        return switch (valor.toLowerCase()) {
            case "usuario" -> TipoEntidadImagen.USUARIO;
            case "evento" -> TipoEntidadImagen.EVENTO;
            case "edicion" -> TipoEntidadImagen.EDICION;
            case "institucion" -> TipoEntidadImagen.INSTITUCION;
            default -> null;
        };
    }
}

package web;

import contratos.dto.DtImagen;
import jakarta.servlet.ServletException;
import jakarta.servlet.http.HttpServletRequest;
import jakarta.servlet.http.Part;

import java.io.IOException;
import java.io.InputStream;

final class Subidas {

    private static final long TAMANIO_MAXIMO = 2L * 1024 * 1024;

    private Subidas() {
    }

    // Devuelve null si no se eligió ningún archivo.
    static DtImagen leerImagen(HttpServletRequest req, String campo)
            throws IOException, ServletException {

        Part parte = req.getPart(campo);

        if (parte == null || parte.getSize() == 0) {
            return null;
        }

        if (parte.getSize() > TAMANIO_MAXIMO) {
            throw new IllegalArgumentException("La imagen supera el tamaño máximo de 2 MB.");
        }

        try (InputStream entrada = parte.getInputStream()) {
            byte[] contenido = entrada.readNBytes((int) TAMANIO_MAXIMO + 1);
            if (contenido.length > TAMANIO_MAXIMO) {
                throw new IllegalArgumentException("La imagen supera el tamaño máximo de 2 MB.");
            }
            return new DtImagen(parte.getContentType(), contenido);
        }
    }
}

        comboEdiciones.removeAllItems();
        String eventoSel = (String) comboEventos.getSelectedItem();
        if (eventoSel != null) {
            Collection<DtEdicion> ediciones = sistema.obtenerEdicionesEvento(eventoSel);

        Collection<DtEdicion> ediciones = sistema.obtenerEdicionesEvento(eventoSel);

        panelForm.add(comboEdiciones);

        panelForm.add(new JLabel("Tipo de Registro:"));
        comboTiposRegistro = new JComboBox<>();
        panelForm.add(comboTiposRegistro);

        panelForm.add(new JLabel("Código de patrocinio (opcional):"));
        txtCodigoPatrocinio = new JTextField();
        panelForm.add(txtCodigoPatrocinio);

        btnRegistrar = new JButton("Confirmar Registro");

        JPanel panelBoton = new JPanel(
                new FlowLayout(FlowLayout.RIGHT)
        );

        panelBoton.add(btnRegistrar);

        add(panelForm, BorderLayout.CENTER);
        add(panelBoton, BorderLayout.SOUTH);

    private void configurarEventos() {
        comboEventos.addActionListener(e -> cargarEdiciones());
        comboEdiciones.addActionListener(e -> cargarTiposRegistro());
        btnRegistrar.addActionListener(e -> ejecutarRegistro());
    }

    public void refrescarDatos() {
        cargarAsistentes();
        cargarEventos();

        comboEdiciones.removeAllItems();
        comboTiposRegistro.removeAllItems();

        comboAsistentes.setSelectedIndex(-1);
        comboEventos.setSelectedIndex(-1);
        comboEdiciones.setSelectedIndex(-1);
        comboTiposRegistro.setSelectedIndex(-1);

        txtCodigoPatrocinio.setText("");
    }

    public void cargarAsistentes() {
        comboAsistentes.removeAllItems();

        Collection<DtUsuario> usuarios = sistema.listarUsuarios();

        for (DtUsuario usuario : usuarios) {
            if (usuario instanceof DtAsistente) {
                comboAsistentes.addItem(usuario.getNickname());
            }
        }

        comboAsistentes.setSelectedIndex(-1);
    }

    public void cargarEventos() {
        comboEventos.removeAllItems();

        Collection<DtEvento> eventos = sistema.listarEventos();

        for (DtEvento evento : eventos) {
            comboEventos.addItem(evento.getNombre());
        }

        comboEventos.setSelectedIndex(-1);
    }

    private void cargarEdiciones() {
        comboEdiciones.removeAllItems();
        comboTiposRegistro.removeAllItems();

        String eventoSeleccionado =
                (String) comboEventos.getSelectedItem();

        if (eventoSeleccionado == null) {
            comboEdiciones.setSelectedIndex(-1);
            comboTiposRegistro.setSelectedIndex(-1);
            return;
        }

        // Solo se puede registrar a ediciones Aceptadas
        Collection<DtEdicion> ediciones =
                sistema.obtenerEdicionesEvento(eventoSeleccionado, EstadoEdicion.ACEPTADA);

        for (DtEdicion edicion : ediciones) {
            comboEdiciones.addItem(edicion.getIdNombre());
        }

        comboEdiciones.setSelectedIndex(-1);
        comboTiposRegistro.setSelectedIndex(-1);
    }

    private void cargarTiposRegistro() {
        comboTiposRegistro.removeAllItems();

        String edicionSeleccionada =
                (String) comboEdiciones.getSelectedItem();

        if (edicionSeleccionada == null) {
            comboTiposRegistro.setSelectedIndex(-1);
            return;
        }

        Collection<DtTipoRegistro> tipos =
                sistema.obtenerTiposRegistroEdicion(edicionSeleccionada);

        for (DtTipoRegistro tipo : tipos) {
            comboTiposRegistro.addItem(tipo.getIdNombre());
        }

        comboTiposRegistro.setSelectedIndex(-1);
    }

    private void ejecutarRegistro() {
        String nickname =
                (String) comboAsistentes.getSelectedItem();

        String eventoSeleccionado =
                (String) comboEventos.getSelectedItem();

        String edicionSeleccionada =
                (String) comboEdiciones.getSelectedItem();

        String tipoRegistroSeleccionado =
                (String) comboTiposRegistro.getSelectedItem();

        if (nickname == null
                || eventoSeleccionado == null
                || edicionSeleccionada == null
                || tipoRegistroSeleccionado == null) {

            JOptionPane.showMessageDialog(
                    this,
                    "Debe seleccionar un asistente, un evento, una edición y un tipo de registro.",
                    "Registro a Edición",
                    JOptionPane.WARNING_MESSAGE
            );
            return;
        }

        try {
            String codigoPatrocinio =
                    txtCodigoPatrocinio.getText().trim();

            DtRegistro dtRegistro = new DtRegistro(
                    LocalDate.now(),
                    0,
                    tipoRegistroSeleccionado,
                    edicionSeleccionada
            );

            sistema.registroAEdicion(
                    nickname,
                    edicionSeleccionada,
                    tipoRegistroSeleccionado,
                    dtRegistro,
                    codigoPatrocinio
            , datatypes.Actor.administradorLocal());

            JOptionPane.showMessageDialog(
                    this,
                    "Registro realizado correctamente.",
                    "Registro a Edición",
                    JOptionPane.INFORMATION_MESSAGE
            );

            refrescarDatos();

        } catch (IllegalArgumentException e) {
            JOptionPane.showMessageDialog(
                    this,
                    e.getMessage(),
                    "Registro a Edición",
                    JOptionPane.WARNING_MESSAGE
            );

        } catch (Exception e) {
            JOptionPane.showMessageDialog(
                    this,
                    "Ocurrió un error inesperado:\n" + e.getMessage(),
                    "Error",
                    JOptionPane.ERROR_MESSAGE
            );
        }
    }

@WebServlet("/alta-patrocinio")
public class AltaPatrocinioServlet extends HttpServlet {
    private static final String[] CAMPOS = {"evento", "edicion", "institucion", "nivel",
            "tipoRegistro", "aporte", "registrograt", "codigo"};

    @Override
    protected void doGet(HttpServletRequest req, HttpServletResponse resp) throws ServletException, IOException {
        Map<String, String> form = new HashMap<>();
        // El código y los importes solo se reciben por POST.
        for (String campo : new String[]{"evento", "edicion"}) form.put(campo, texto(req, campo));
        mostrarFormulario(req, resp, form, null);
    }

    @Override
    protected void doPost(HttpServletRequest req, HttpServletResponse resp) throws ServletException, IOException {
        ServicioPortal sistema = ContextoListener.central(req);
        String edicion = texto(req, "edicion");
        Map<String, String> form = new HashMap<>();
        for (String campo : CAMPOS) form.put(campo, texto(req, campo));
        try {
            String evento = form.get("evento");
            float aporte;
            int gratuitos;
            try {
                aporte = Float.parseFloat(form.get("aporte"));
                gratuitos = Integer.parseInt(form.get("registrograt"));
            } catch (NumberFormatException e) {
                throw new IllegalArgumentException("Ingresá un aporte numérico y una cantidad entera de registros gratuitos.");
            }
            NivelPatrocinio nivel;
            try {
                nivel = NivelPatrocinio.valueOf(form.get("nivel"));
            } catch (IllegalArgumentException e) {
                throw new IllegalArgumentException("Seleccioná un nivel de patrocinio válido.");
            }
            DtPatrocinio dt = new DtPatrocinio(LocalDate.now(), aporte, gratuitos, form.get("codigo"),
                    nivel, form.get("institucion"), edicion, form.get("tipoRegistro"));
            sistema.altaPatrocinio(dt, evento);
            resp.sendRedirect(req.getContextPath() + "/consulta-patrocinio?edicion="
                    + URLEncoder.encode(edicion, StandardCharsets.UTF_8) + "&institucion="
                    + URLEncoder.encode(form.get("institucion"), StandardCharsets.UTF_8) + "&exito=true");
        } catch (IllegalArgumentException e) {
            mostrarFormulario(req, resp, form, e.getMessage());
        }
    }

    private void mostrarFormulario(HttpServletRequest req, HttpServletResponse resp,
                                   Map<String, String> form, String error) throws ServletException, IOException {
        ServicioPortal sistema = ContextoListener.central(req);
        String edicion = form.getOrDefault("edicion", "");
        DtOpcionesAltaPatrocinio opciones;
        opciones = sistema.obtenerOpcionesAltaPatrocinio(form.getOrDefault("evento", ""), edicion);
        form.put("evento", opciones.getEvento());
        req.setAttribute("form", form);
        req.setAttribute("error", error);
        req.setAttribute("eventos", opciones.getEventos());
        req.setAttribute("ediciones", opciones.getEdiciones());
        Collection<DtTipoRegistro> tipos = opciones.getTipos();
        Collection<DtInstitucion> instituciones = opciones.getInstituciones();
        req.setAttribute("tipos", tipos);
        req.setAttribute("instituciones", instituciones);
        req.setAttribute("tipoNoDisponible", !form.getOrDefault("tipoRegistro", "").isEmpty()
                && tipos.stream().noneMatch(t -> t.getIdNombre().equals(form.get("tipoRegistro"))));
        req.setAttribute("institucionNoDisponible", !form.getOrDefault("institucion", "").isEmpty()
                && instituciones.stream().noneMatch(i -> i.getNombre().equals(form.get("institucion"))));
        req.setAttribute("nivelNoDisponible", !form.getOrDefault("nivel", "").isEmpty()
                && Arrays.stream(NivelPatrocinio.values()).noneMatch(n -> n.name().equals(form.get("nivel"))));
        req.setAttribute("niveles", NivelPatrocinio.values());
        req.getRequestDispatcher("/WEB-INF/views/alta-patrocinio.jsp").forward(req, resp);
    }

    private String texto(HttpServletRequest req, String campo) {
        String valor = req.getParameter(campo);
        return valor == null ? "" : valor.trim();
    }

            MensajesFlash.guardar(req, "/usuario", "Los datos del perfil se guardaron.", null);
            resp.sendRedirect(req.getContextPath() + "/usuario?nick="
                    + URLEncoder.encode(perfil.getNickname(), StandardCharsets.UTF_8));

        } catch (IllegalArgumentException e) {
            mostrarFormulario(req, resp, perfil, form, e.getMessage());
        } catch (DateTimeParseException e) {
            mostrarFormulario(req, resp, perfil, form, "La fecha de nacimiento no es válida.");
        }
    }

    private DtUsuario perfilActual(HttpServletRequest req) {
        return ContextoListener.central(req).usuarioActual();
    }

    private DtAsistente armarAsistente(HttpServletRequest req, Map<String, String> form, DtUsuario perfil) {

        LocalDate fechaNacimiento = LocalDate.parse(form.get("fechaNac"));


        String institucion = form.get("institucion").isEmpty() ? null : form.get("institucion");

        DtAsistente dt = new DtAsistente(
                perfil.getNickname(), form.get("nombre"), perfil.getCorreoElectronico(),
                form.get("apellido"), fechaNacimiento, institucion);


        return dt;
    }

    private DtOrganizador armarOrganizador(HttpServletRequest req, Map<String, String> form, DtUsuario perfil) {

        DtOrganizador dt = new DtOrganizador(
                perfil.getNickname(), form.get("nombre"), perfil.getCorreoElectronico(),
                form.get("descripcion"), form.get("sitioWeb"));


        return dt;
    }

    private Map<String, String> formularioDesde(DtUsuario perfil) {
        Map<String, String> form = new HashMap<>();
        form.put("nombre", vacio(perfil.getNombre()));

        if (perfil instanceof DtAsistente asistente) {
            form.put("apellido", vacio(asistente.getApellido()));
            form.put("fechaNac", asistente.getFechaNacimiento() == null
                    ? "" : asistente.getFechaNacimiento().toString());
            form.put("institucion", vacio(asistente.getNombreInstitucion()));
        } else if (perfil instanceof DtOrganizador organizador) {
            form.put("descripcion", vacio(organizador.getDescripcion()));
            form.put("sitioWeb", vacio(organizador.getSitioWeb()));
        }
        return form;
    }

    private String vacio(String texto) {
        return texto == null ? "" : texto;
    }

    private void mostrarFormulario(HttpServletRequest req, HttpServletResponse resp,
                                   DtUsuario perfil, Map<String, String> form, String error)
            throws ServletException, IOException {

        req.setAttribute("perfil", perfil);
        req.setAttribute("esAsistente", perfil instanceof DtAsistente);
        req.setAttribute("esOrganizador", perfil instanceof DtOrganizador);
        req.setAttribute("form", form);
        req.setAttribute("error", error);
        req.setAttribute("instituciones", ContextoListener.central(req).listarInstituciones());
        req.getRequestDispatcher("/WEB-INF/views/modificar-usuario.jsp").forward(req, resp);
    }
}

@WebServlet("/usuario")
public class UsuarioServlet extends HttpServlet {

    private static final DateTimeFormatter FECHA = DateTimeFormatter.ofPattern("dd/MM/yyyy");

    @Override
    protected void doGet(HttpServletRequest req, HttpServletResponse resp)
            throws ServletException, IOException {

        String nick = req.getParameter("nick");
        ServicioPortal sistema = ContextoListener.central(req);

        var consulta=sistema.consultarPerfil(nick);
        DtUsuario perfil=consulta.getUsuario();boolean esPropio=consulta.isPropio();
        req.setAttribute("fechaAlta", perfil.getFechaAlta() == null
                ? "No registrada" : perfil.getFechaAlta().format(FECHA));
        req.setAttribute("perfil", perfil);
        req.setAttribute("esPropio", esPropio);

        if (perfil instanceof DtAsistente asistente) {
            req.setAttribute("esAsistente", true);
            req.setAttribute("fechaNacimiento", asistente.getFechaNacimiento() == null
                    ? "" : asistente.getFechaNacimiento().format(FECHA));

            if (esPropio) {
                Collection<RegistroVista> registros = new ArrayList<>();

                for (DtRegistro registro : consulta.getRegistros()) {
                    float costo = registro.getCosto();

                    registros.add(new RegistroVista(
                            registro.getNombreEdicion(),
                            registro.getNombreTipoRegistro(),
                            costo == Math.rint(costo) ? String.valueOf((long) costo) : String.format("%.2f", costo),
                            registro.getFechaRegistro() == null ? "" : registro.getFechaRegistro().format(FECHA)));
                }

                req.setAttribute("registros", registros);
            }
        } else if (perfil instanceof DtOrganizador) {
            req.setAttribute("esOrganizador", true);

            var ediciones=consulta.getEdiciones();
            req.setAttribute("ediciones", ediciones);
        }

        if (esPropio) MensajesFlash.mostrar(req, "/usuario");
        req.getRequestDispatcher("/WEB-INF/views/usuario.jsp").forward(req, resp);
    }
}

package swing;

import datatypes.Actor;

import javax.swing.*;

public class AltaEdicionEventoPanel extends JPanel {

    private final ISistema sistema;
    private final ImagenSwing imagen = new ImagenSwing(true);

    private JComboBox<String> comboEventos;
    private JComboBox<String> comboOrganizadores;

    private JTextField txtNombreEdicion;
    private JTextField txtSigla;
    private JSpinner spinnerFechaInicio;
    private JSpinner spinnerFechaFin;
    private JTextField txtCiudad;
    private JTextField txtPais;

    private final Actor actor;

    public AltaEdicionEventoPanel(Actor actor) {
        this(actor, Fabrica.getInstance().getISistema());
    }

    AltaEdicionEventoPanel(Actor actor, ISistema sistema) {
        this.actor = java.util.Objects.requireNonNull(actor, "Se requiere Actor.");

        this.sistema = java.util.Objects.requireNonNull(sistema);

        initUI();

        cargarEventos();
        cargarOrganizadores();
    }

    private void initUI() {

        setLayout(new BorderLayout(10, 10));

        // =====================================================
        // DATOS DE LA EDICIÓN
        // =====================================================

        JPanel panelForm = new JPanel(new GridLayout(8, 2, 5, 5));

        panelForm.setBorder(
                BorderFactory.createTitledBorder(
                        "Datos de la Edición"
                )
        );

        panelForm.add(new JLabel("Seleccionar Evento:"));
        comboEventos = new JComboBox<>();
        panelForm.add(comboEventos);

        panelForm.add(new JLabel("Organizador:"));
        comboOrganizadores = new JComboBox<>();
        panelForm.add(comboOrganizadores);

        panelForm.add(new JLabel("Nombre / ID Edición:"));
        txtNombreEdicion = new JTextField();
        panelForm.add(txtNombreEdicion);

        panelForm.add(new JLabel("Sigla:"));
        txtSigla = new JTextField();
        panelForm.add(txtSigla);

        panelForm.add(new JLabel("Fecha Inicio (AAAA-MM-DD):"));

        spinnerFechaInicio.setEditor(
                new JSpinner.DateEditor(
                        spinnerFechaInicio,
                        "yyyy-MM-dd"
                )
        );
        panelForm.add(spinnerFechaInicio);

        panelForm.add(new JLabel("Fecha Fin (AAAA-MM-DD):"));
        spinnerFechaFin = new JSpinner(new SpinnerDateModel());
        spinnerFechaFin.setEditor(
                new JSpinner.DateEditor(
                        spinnerFechaFin,
                        "yyyy-MM-dd"
                )
        );
        panelForm.add(spinnerFechaFin);

        panelForm.add(new JLabel("Ciudad:"));
        txtCiudad = new JTextField();
        panelForm.add(txtCiudad);

        panelForm.add(new JLabel("País:"));
        txtPais = new JTextField();
        panelForm.add(txtPais);

        // =====================================================
        // BOTÓN GUARDAR
        // =====================================================

        JButton btnGuardar = new JButton("Confirmar");
        btnGuardar.setFont(
                new Font(
                        "SansSerif",
                        Font.BOLD,
                        14
                )
        );

        btnGuardar.addActionListener(
                e -> guardarEdicion()
        );

        JPanel contenido = new JPanel(new BorderLayout(8, 8));
        contenido.add(panelForm, BorderLayout.NORTH);
        contenido.add(imagen, BorderLayout.CENTER);
        add(new JScrollPane(contenido), BorderLayout.CENTER);
        add(btnGuardar, BorderLayout.SOUTH);
    }

    // =====================================================
    // CARGAR EVENTOS
    // =====================================================

    public void cargarEventos() {

        comboEventos.removeAllItems();

        Collection<DtEvento> eventos = sistema.listarEventos();

        for (DtEvento evento : eventos) {
            comboEventos.addItem(
                    evento.getNombre()
            );
        }
    }

    // =====================================================
    // CARGAR ORGANIZADORES
    // =====================================================

    public void cargarOrganizadores() {

        comboOrganizadores.removeAllItems();

        Collection<DtUsuario> usuarios = sistema.listarUsuarios();

        for (DtUsuario usuario : usuarios) {

            if (usuario instanceof DtOrganizador) {

                comboOrganizadores.addItem(
                        usuario.getNickname()
                );
            }
        }
    }

    // =====================================================
    // GUARDAR EDICIÓN
    // =====================================================

    private void guardarEdicion() {

        String eventoSel = (String) comboEventos.getSelectedItem();
        String organizadorSel = (String) comboOrganizadores.getSelectedItem();
        String nombre = txtNombreEdicion.getText().trim();
        String sigla = txtSigla.getText().trim();
        String ciudad = txtCiudad.getText().trim();
        String pais = txtPais.getText().trim();

        if (eventoSel == null
                || organizadorSel == null
                || nombre.isEmpty()
                || sigla.isEmpty()
                || ciudad.isEmpty()
                || pais.isEmpty()) {

            JOptionPane.showMessageDialog(
                    this,
                    "Por favor complete todos los campos de la edición.",
                    "Atención",
                    JOptionPane.WARNING_MESSAGE
            );

            return;
        }

        LocalDate fInicio = toLocalDate((Date) spinnerFechaInicio.getValue());
        LocalDate fFin = toLocalDate((Date) spinnerFechaFin.getValue());

        if (fFin.isBefore(fInicio)) {

            JOptionPane.showMessageDialog(
                    this,
                    "La fecha de fin no puede ser anterior a la de inicio.",
                    "Error",
                    JOptionPane.ERROR_MESSAGE
            );

            return;
        }

        DtEdicion dtEdicion = new DtEdicion(
                nombre,
                sigla,
                fInicio,
                fFin,
                LocalDate.now(),
                ciudad,
                pais,
                organizadorSel
        );

        try {

            DtImagen nuevaImagen = imagen.leerValidada(sistema);
            boolean okEdicion = sistema.altaEdicion(dtEdicion, eventoSel, nuevaImagen, actor);

            if (okEdicion) {

                JOptionPane.showMessageDialog(
                        this,
                        "Edición registrada correctamente.",
                        "Éxito",
                        JOptionPane.INFORMATION_MESSAGE
                );

                limpiarFormulario();

                JInternalFrame internalFrame = (JInternalFrame) SwingUtilities.getAncestorOfClass(JInternalFrame.class, this);
                if (internalFrame != null) {
                    internalFrame.setVisible(false); // Oculta la ventana sin destruirla
                }

            } else {

                JOptionPane.showMessageDialog(
                        this,
                        "No se pudo crear la edición. Verifique que el nombre no exista y que el organizador sea válido.",
                        "Error",
                        JOptionPane.ERROR_MESSAGE
                );
            }

        } catch (Exception ex) {

            JOptionPane.showMessageDialog(
                    this,
                    "Error: " + ex.getMessage(),
                    "Error",
                    JOptionPane.ERROR_MESSAGE
            );
        }
    }

    private LocalDate toLocalDate(Date date) {

        return date.toInstant()
                .atZone(ZoneId.systemDefault())
                .toLocalDate();
    }

    private void limpiarFormulario() {
        imagen.mostrar(null);

        txtNombreEdicion.setText("");
        txtSigla.setText("");
        txtCiudad.setText("");
        txtPais.setText("");
    }
}

package web;

import contratos.dto.*;
import contratos.ServicioPortal;
import jakarta.servlet.ServletException;
import jakarta.servlet.annotation.WebServlet;
import jakarta.servlet.http.*;

import java.io.IOException;
import java.util.Collection;
import java.util.HashMap;
import java.util.Map;

@WebServlet("/alta-tipo-registro")
public class AltaTipoRegistroServlet extends HttpServlet {

    @Override
    protected void doGet(HttpServletRequest req, HttpServletResponse resp)
            throws ServletException, IOException {


        MensajesFlash.mostrar(req, "/alta-tipo-registro");

        ServicioPortal sistema = ContextoListener.central(req);

        Collection<DtEdicion> ediciones =
                sistema.consultarPerfil(SesionWeb.usuario(req) == null ? "" : SesionWeb.usuario(req).getNickname()).getEdiciones();

        req.setAttribute("ediciones", ediciones);

        req.getRequestDispatcher(
                "/WEB-INF/views/alta-tipo-registro.jsp"
        ).forward(req, resp);
    }

    @Override
    protected void doPost(HttpServletRequest req, HttpServletResponse resp)
            throws ServletException, IOException {


        ServicioPortal sistema = ContextoListener.central(req);

        Map<String, String> form = new HashMap<>();
        for (String campo : new String[]{"edicion", "nombre", "descripcion", "costo", "cupo"}) {
            String valor = req.getParameter(campo);
            form.put(campo, valor == null ? "" : valor);
        }
        req.setAttribute("form", form);

        try {
            String edicion = form.get("edicion").trim();
            String nombre = form.get("nombre").trim();
            String descripcion = form.get("descripcion").trim();
            float costo = Float.parseFloat(campoNumericoRequerido(form, "costo"));
            int cupo = Integer.parseInt(campoNumericoRequerido(form, "cupo"));

            DtTipoRegistro tipo = new DtTipoRegistro(
                    nombre,
                    descripcion,
                    costo,
                    cupo
            );

            boolean creado = sistema.altaTipoRegistro(
                    tipo,
                    edicion
            );

            if (!creado) {
                throw new IllegalArgumentException(
                        "No se pudo crear el tipo de registro. " +
                                "Verificá que no exista otro con el mismo nombre."
                );
            }

            MensajesFlash.guardar(req, "/alta-tipo-registro", "Tipo de registro creado correctamente.", null);
            resp.sendRedirect(req.getContextPath() + "/alta-tipo-registro");
            return;

        } catch (NumberFormatException e) {
            req.setAttribute(
                    "error",
                    "El costo y el cupo deben ser números válidos."
            );

        } catch (IllegalArgumentException e) {
            req.setAttribute("error", e.getMessage());
        }

        doGet(req, resp);
    }

    private String campoNumericoRequerido(Map<String, String> form, String campo) {
        String valor = form.get(campo);
        if (valor.isBlank()) {
            throw new IllegalArgumentException("El campo " + campo + " es obligatorio.");
        }
        return valor.trim();
    }

}

package web;

import contratos.dto.DtOrganizador;
import contratos.dto.DtUsuario;
import jakarta.servlet.Filter;
import jakarta.servlet.FilterChain;
import jakarta.servlet.ServletException;
import jakarta.servlet.ServletRequest;
import jakarta.servlet.ServletResponse;
import jakarta.servlet.annotation.WebFilter;
import jakarta.servlet.http.HttpServletRequest;
import jakarta.servlet.http.HttpServletResponse;
import jakarta.servlet.http.HttpSession;

import java.io.IOException;
import java.util.Map;
import java.util.Set;

@WebFilter("/*")
public class FiltroAcceso implements Filter {

    private static final Set<String> SOLO_VISITANTE = Set.of("visitante");
    private static final Set<String> SOLO_ASISTENTE = Set.of("asistente");
    private static final Set<String> SOLO_ORGANIZADOR = Set.of("organizador");
    private static final Set<String> CON_SESION = Set.of("asistente", "organizador");

    // Rutas restringidas por rol. Las que no figuran son públicas.
    private static final Map<String, Set<String>> PERMITIDOS = Map.ofEntries(
            Map.entry("/login", SOLO_VISITANTE),
            Map.entry("/alta-usuario", SOLO_VISITANTE),
            Map.entry("/logout", CON_SESION),
            Map.entry("/modificar-usuario", CON_SESION),
            Map.entry("/consulta-registro", CON_SESION),
            Map.entry("/registro-edicion", SOLO_ASISTENTE),
            Map.entry("/alta-evento", SOLO_ORGANIZADOR),
            Map.entry("/alta-edicion", SOLO_ORGANIZADOR),
            Map.entry("/alta-tipo-registro", SOLO_ORGANIZADOR),
            Map.entry("/alta-institucion", SOLO_ORGANIZADOR),
            Map.entry("/alta-patrocinio", SOLO_ORGANIZADOR)
    );

    private static final String[] ESTATICOS = {"/css/", "/js/", "/vendor/", "/img/"};

    @Override
    public void doFilter(ServletRequest request, ServletResponse response, FilterChain chain)
            throws IOException, ServletException {

        HttpServletRequest req = (HttpServletRequest) request;
        HttpServletResponse resp = (HttpServletResponse) response;

        try {
            filtrar(req, resp, chain);
        } catch (web.remoto.ErrorCentral error) {
            responderError(req, resp, error);
        } catch (IllegalArgumentException error) {
            resp.sendError(400, "La solicitud no es válida.");
        }
    }

    static void responderError(HttpServletRequest req, HttpServletResponse resp, web.remoto.ErrorCentral error)
            throws IOException, ServletException {
            switch (error.getCodigo()) {
                case "SESION_INVALIDA" -> {
                    var sesion = req.getSession(false);
                    if (sesion != null) sesion.invalidate();
                    resp.sendRedirect(req.getContextPath() + "/login");
                }
                case "ACCESO_DENEGADO" -> resp.sendError(403);
                case "NO_ENCONTRADO" -> resp.sendError(404);
                default -> {
                    resp.setStatus(503);
                    req.getRequestDispatcher("/WEB-INF/views/servicio-no-disponible.jsp").forward(req, resp);
                }
            }
    }

    private void filtrar(HttpServletRequest req, HttpServletResponse resp, FilterChain chain)
            throws IOException, ServletException {

        req.setCharacterEncoding("UTF-8");

        String ruta = req.getServletPath();

        if ("POST".equals(req.getMethod()) && !Csrf.validarPost(req, resp)) return;

        // La consulta AJAX pública no necesita cabezal, categorías ni una sesión HTTP.
        // Su Servlet conserva la respuesta JSON incluso si Central no está disponible.
        if (esEstatico(ruta) || "/disponibilidad-usuario".equals(ruta)) {
            chain.doFilter(req, resp);
            return;
        }

        if (!"POST".equals(req.getMethod())) Csrf.prepararFormulario(req);
        resp.setHeader("Cache-Control", "private, no-store");

        HttpSession sesion = req.getSession(false);
        DtUsuario usuario = SesionWeb.usuario(req);
        if (!"/logout".equals(ruta) && SesionWeb.credencial(req) != null) {
            usuario = ContextoListener.central(req).usuarioActual();
            SesionWeb.guardar(sesion, usuario);
        } else if (usuario != null && SesionWeb.credencial(req) == null) {
            sesion.removeAttribute("usuario");
            usuario = null;
        }
        String rol = rolDe(usuario);

        Set<String> permitidos = PERMITIDOS.get(ruta);

        if (permitidos != null && !permitidos.contains(rol)) {
            if (rol.equals("visitante")) {
                resp.sendRedirect(req.getContextPath() + "/login");
            } else if (permitidos.equals(SOLO_VISITANTE)) {
                resp.sendRedirect(req.getContextPath() + "/");
            } else {
                resp.sendError(HttpServletResponse.SC_FORBIDDEN);
            }
            return;
        }

        // Datos que necesita el cabezal en todas las páginas
        req.setAttribute("rol", rol);
        req.setAttribute("categorias", ContextoListener.central(req).listarCategorias());

        chain.doFilter(req, resp);
    }

    private boolean esEstatico(String ruta) {
        if (ruta.equals("/imagen")) {
            return true;
        }
        for (String prefijo : ESTATICOS) {
            if (ruta.startsWith(prefijo)) {
                return true;
            }
        }
        return false;
    }

    private String rolDe(DtUsuario usuario) {
        if (usuario == null) {
            return "visitante";
        }
        return usuario instanceof DtOrganizador ? "organizador" : "asistente";
    }
}
